# LiteRGSS — Ruby bindings over LiteCGSS.
#
# Boundary: the only way into the engine is LiteCGSS::engine, which exposes
# the public litecgss/include/ tree (as <LiteCGSS/...>) plus the
# configuration defines it was compiled with. Never add litecgss/src (or any
# other engine path) to this target's include directories.

include(RubyExtension)

file(GLOB_RECURSE LITERGSS_SOURCES CONFIGURE_DEPENDS
	${CMAKE_CURRENT_SOURCE_DIR}/*.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/*.h
)

add_ruby_extension(LiteRGSS ${LITERGSS_SOURCES})

if(LITERGSS_EMBEDDED)
	# A static library a host links (litergss-everywhere's runtime): it calls
	# Init_LiteRGSS itself, in every build (Tracy included), and installs it.
	set(LITERGSS_EXT_NAME LiteRGSS)
	install(TARGETS LiteRGSS ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
else()
	# Presets write it to build/<preset>/lib (CMakePresets.json), so builds of
	# different backends / options never overwrite each other; build/lib is for
	# builds configured without a preset.
	set(LITERGSS_EXT_OUTPUT_DIR ${PROJECT_SOURCE_DIR}/build/lib CACHE PATH
		"Where the LiteRGSS extension is written")
	set_target_properties(LiteRGSS PROPERTIES
		LIBRARY_OUTPUT_DIRECTORY ${LITERGSS_EXT_OUTPUT_DIR}
		RUNTIME_OUTPUT_DIRECTORY ${LITERGSS_EXT_OUTPUT_DIR}
	)

	# A Tracy build needs the host to start the profiler before any LiteRGSS
	# call, so it is a different extension (LiteRGSS_tracy, as in LiteRGSS2),
	# loaded with the shared TracyClient (of the Tracy package) linked next to it.
	if(LITECGSS_USE_TRACY)
		set(LITERGSS_EXT_NAME LiteRGSS_tracy)
		add_custom_command(TARGET LiteRGSS POST_BUILD
			COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE:Tracy::TracyClient>
				${LITERGSS_EXT_OUTPUT_DIR}/$<TARGET_FILE_NAME:Tracy::TracyClient>
			VERBATIM)
		if(APPLE)
			set_target_properties(LiteRGSS PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
		elseif(UNIX)
			set_target_properties(LiteRGSS PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
		endif()
	else()
		set(LITERGSS_EXT_NAME LiteRGSS)
	endif()
	set_target_properties(LiteRGSS PROPERTIES OUTPUT_NAME ${LITERGSS_EXT_NAME})
	# tools/ (the agent CLIs) bake these in.
	set(LITERGSS_EXT_NAME ${LITERGSS_EXT_NAME} PARENT_SCOPE)
	set(LITERGSS_EXT_OUTPUT_DIR ${LITERGSS_EXT_OUTPUT_DIR} PARENT_SCOPE)

	# PSDK's Unix loader requires "$GAMEDEPS/ruby-dist/lib/LiteRGSS": every build
	# stages its extension there (a symlink, so a rebuild is picked up), for
	# tools/psdk-demo.sh and litergss-run.
	set(LITERGSS_GAMEDEPS_DIR ${CMAKE_BINARY_DIR}/psdk-gamedeps)
	add_custom_command(TARGET LiteRGSS POST_BUILD
		COMMAND ${CMAKE_COMMAND} -E make_directory ${LITERGSS_GAMEDEPS_DIR}/ruby-dist/lib
		COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE:LiteRGSS>
			${LITERGSS_GAMEDEPS_DIR}/ruby-dist/lib/$<TARGET_FILE_NAME:LiteRGSS>
		VERBATIM)
	if(LITECGSS_USE_TRACY)
		# PSDK requires "LiteRGSS": a shim loads LiteRGSS_tracy, next to the
		# TracyClient it links ($ORIGIN is the staged folder).
		file(WRITE ${LITERGSS_GAMEDEPS_DIR}/ruby-dist/lib/LiteRGSS.rb
			"# Generated by ext/LiteRGSS3/CMakeLists.txt: PSDK requires LiteRGSS; this build is LiteRGSS_tracy.\nrequire_relative 'LiteRGSS_tracy'\n")
		add_custom_command(TARGET LiteRGSS POST_BUILD
			COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE:Tracy::TracyClient>
				${LITERGSS_GAMEDEPS_DIR}/ruby-dist/lib/$<TARGET_FILE_NAME:Tracy::TracyClient>
			VERBATIM)
	endif()
	set(LITERGSS_GAMEDEPS_DIR ${LITERGSS_GAMEDEPS_DIR} PARENT_SCOPE)
endif()
target_compile_definitions(LiteRGSS PRIVATE LITERGSS_INIT_FUNCTION=Init_${LITERGSS_EXT_NAME})

# Bindings include their own headers by bare name ("Color.h") from any
# subfolder: every subfolder is an include root.
file(GLOB _litergss_children LIST_DIRECTORIES true ${CMAKE_CURRENT_SOURCE_DIR}/*)
set(_litergss_include_dirs ${CMAKE_CURRENT_SOURCE_DIR})
foreach(_child IN LISTS _litergss_children)
	if(IS_DIRECTORY ${_child})
		list(APPEND _litergss_include_dirs ${_child})
	endif()
endforeach()
target_include_directories(LiteRGSS PRIVATE ${_litergss_include_dirs})

target_link_libraries(LiteRGSS PRIVATE LiteCGSS::engine)

# The agent runtime (agent/agent_runtime.rb) is compiled into the extension:
# LiteRGSS::Agent works in any host, with no file to ship next to it.
set(_agent_runtime ${CMAKE_CURRENT_SOURCE_DIR}/agent/agent_runtime.rb)
set(_agent_runtime_header ${CMAKE_CURRENT_BINARY_DIR}/generated/AgentRuntime.h)
add_custom_command(OUTPUT ${_agent_runtime_header}
	COMMAND ${CMAKE_COMMAND} -DINPUT=${_agent_runtime} -DOUTPUT=${_agent_runtime_header} -DNAME=kAgentRuntime
		-P ${PROJECT_SOURCE_DIR}/cmake/EmbedText.cmake
	DEPENDS ${_agent_runtime} ${PROJECT_SOURCE_DIR}/cmake/EmbedText.cmake
	COMMENT "Embedding the LiteRGSS agent runtime"
	VERBATIM)
target_sources(LiteRGSS PRIVATE ${_agent_runtime_header})
target_include_directories(LiteRGSS PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)

if(NOT LITERGSS_EMBEDDED)
	# Ruby test suite (non-interactive files only: *_live.rb open a real window
	# and wait for user input). One CTest entry per file.
	# Not in a Tracy build: its extension needs a host starting the profiler
	# first (tracy_ruby), which the suite has no way to do.
	file(GLOB _litergss_ruby_tests ${PROJECT_SOURCE_DIR}/tests/test_*.rb)
	if(LITECGSS_USE_TRACY)
		set(_litergss_ruby_tests "")
	endif()
	list(FILTER _litergss_ruby_tests EXCLUDE REGEX "_live\\.rb$")
	foreach(_test IN LISTS _litergss_ruby_tests)
		get_filename_component(_name ${_test} NAME_WE)
		add_test(NAME LiteRGSS.${_name}
			COMMAND ${RUBY_EXECUTABLE} -Itests ${_test}
			WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
		# The extension this build produced (tests/litergss.rb loads it), its
		# windows hidden: the suite doesn't open windows on the desktop
		set_tests_properties(LiteRGSS.${_name} PROPERTIES
			ENVIRONMENT "LITERGSS_LIB=${LITERGSS_EXT_OUTPUT_DIR}/LiteRGSS;LITERGSS_HIDDEN_WINDOW=1")
	endforeach()
endif()
